\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=15cm,
  height=9cm,
  xmin=-5,
  xmax=3,
  ymin=-1,
  ymax=12,
  axis lines=middle,
  xlabel={$x$},
  ylabel={$y$},
  grid=both,
  minor tick num=1,
  major grid style={draw=gray!30},
  minor grid style={draw=gray!14},

  clip=false

\node[complexorange,font=\bfseries,anchor=south west]
  at (axis cs:-1,4) {vertex $(-1,4)$};
